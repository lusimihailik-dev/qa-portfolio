📁 Навигация и файлы
Команда	Описание

mkdir	создать папку
rmdir	удалить пустую папку

cp	копировать (cp -r — рекурсивно)
mv	переместить/переименовать
rm	удалить (rm -rf — принудительно)
find	поиск файлов
locate	быстрый поиск по базе
tree	дерево каталогов
